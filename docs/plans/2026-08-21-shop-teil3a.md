const VIEWER_EXPIRES_IN = '7d';

/**
 * Shop-Ausweis fuer einen Zuschauer. Bewusst eine ANDERE Form als der
 * Streamer-Ausweis: `typ: 'viewer'` trennt beide Welten hart voneinander.
 */
function signViewerToken({ viewerId, twitchId }) {
    return jwt.sign({ viewerId, twitchId, typ: 'viewer' }, JWT_SECRET, { expiresIn: VIEWER_EXPIRES_IN });
}

/** Guard fuer Shop-Endpunkte: akzeptiert NUR Zuschauer-Ausweise. */
function authenticateViewer(req, res, next) {
    const authHeader = req.headers.authorization || req.headers['authorization'];
    if (!authHeader || !authHeader.startsWith('Bearer ')) {
        return res.status(401).json({ error: 'Unauthorized' });
    }
    let decoded;
    try {
        decoded = jwt.verify(authHeader.split(' ')[1], JWT_SECRET);
    } catch (err) {
        return res.status(401).json({ error: err.name === 'TokenExpiredError' ? 'Token expired' : 'Invalid token' });
    }
    if (decoded.typ !== 'viewer' || !decoded.viewerId) {
        return res.status(401).json({ error: 'Unauthorized' });
    }
    req.viewer = { id: decoded.viewerId, twitchId: String(decoded.twitchId || '') };
    next();
}
```

/**
 * Shop-Endpunkte fuer Zuschauer (Teil 3a).
 *
 * Die Shop-Seite ist statisch und haelt KEINE Supabase-Zugangsdaten. Sie
 * meldet sich mit einem Twitch-Token an, den wir hier bei Twitch pruefen,
 * und bekommt dafuer einen Shop-Ausweis (siehe middleware/auth.js).
 */
const express = require('express');
const { createClient } = require('@supabase/supabase-js');
const { signViewerToken, authenticateViewer } = require('../middleware/auth');
const { checkSpeechText } = require('../lib/speechFilter');

const router = express.Router();
const supabase = createClient(process.env.SUPABASE_URL, process.env.SUPABASE_SERVICE_ROLE_KEY);

// Muss zur Twitch-Anwendung passen, mit der sich die Seite anmeldet.
const TWITCH_CLIENT_ID = process.env.TWITCH_CLIENT_ID || '5ns5vekvgz8wb6wudfsos1nvzsa4sq';

/**
 * GET /api/shop/catalog — oeffentlich, keine Anmeldung noetig.
 */
router.get('/catalog', async (req, res) => {
    try {
        const { data, error } = await supabase
            .from('skins')
            .select('key, name, kind, price_cents, tint_hex, duration_days')
            .eq('active', true)
            .order('kind')
            .order('price_cents');
        if (error) throw error;
        res.json({ items: data || [] });
    } catch (err) {
        console.error('[shop] catalog:', err.message);
        res.status(500).json({ error: 'catalog_unavailable' });
    }
});

/**
 * POST /api/shop/login  { twitchAccessToken }
 * Prueft den Token BEI TWITCH (nicht dem Client vertrauen), legt den
 * Zuschauer an und stellt den Shop-Ausweis aus.
 */
router.post('/login', async (req, res) => {
    const token = req.body?.twitchAccessToken;
    if (!token || typeof token !== 'string') {
        return res.status(400).json({ error: 'twitchAccessToken erforderlich' });
    }

    try {
        const vRes = await fetch('https://id.twitch.tv/oauth2/validate', {
            headers: { Authorization: `OAuth ${token}` },
            signal: AbortSignal.timeout(8000)
        });
        if (!vRes.ok) return res.status(401).json({ error: 'twitch_invalid' });
        const v = await vRes.json();

        // Audience-Bindung: der Token muss fuer UNSERE Anwendung ausgestellt sein.
        if (v.client_id !== TWITCH_CLIENT_ID) {
            console.warn('[shop] login: fremde client_id', v.client_id);
            return res.status(401).json({ error: 'twitch_invalid' });
        }

        const uRes = await fetch('https://api.twitch.tv/helix/users', {
            headers: { 'Client-Id': TWITCH_CLIENT_ID, Authorization: `Bearer ${token}` },
            signal: AbortSignal.timeout(8000)
        });
        const twitchUser = (await uRes.json())?.data?.[0];
        if (!twitchUser?.id) return res.status(401).json({ error: 'twitch_invalid' });

        const { data: viewer, error } = await supabase
            .from('viewers')
            .upsert({
                twitch_id: String(twitchUser.id),
                twitch_login: twitchUser.login,
                display_name: twitchUser.display_name || twitchUser.login
            }, { onConflict: 'twitch_id' })
            .select('id, twitch_id, display_name')
            .single();
        if (error || !viewer) throw error || new Error('viewer upsert leer');

        res.json({
            token: signViewerToken({ viewerId: viewer.id, twitchId: viewer.twitch_id }),
            viewer: { displayName: viewer.display_name, twitchId: viewer.twitch_id }
        });
    } catch (err) {
        console.error('[shop] login:', err.message);
        res.status(500).json({ error: 'login_failed' });
    }
});

module.exports = router;
```

```js
/**
 * GET /api/shop/me — Besitz, aktiver Skin, eigener Blasen-Text.
 * Abgelaufene Eintraege werden mitgeliefert und als abgelaufen markiert,
 * damit der Zuschauer sieht, dass er nachkaufen kann.
 */
router.get('/me', authenticateViewer, async (req, res) => {
    try {
        const { data: besitz, error } = await supabase
            .from('viewer_skins')
            .select('equipped, expires_at, revoked_at, skins!inner(key, name, kind, tint_hex, duration_days)')
            .eq('viewer_id', req.viewer.id)
            .is('revoked_at', null);
        if (error) throw error;

        const { data: blase } = await supabase
            .from('viewer_speech')
            .select('text, status, updated_at')
            .eq('viewer_id', req.viewer.id)
            .maybeSingle();

        const jetzt = Date.now();
        res.json({
            items: (besitz || []).map(z => ({
                key: z.skins.key,
                name: z.skins.name,
                kind: z.skins.kind,
                tint_hex: z.skins.tint_hex,
                equipped: !!z.equipped,
                expires_at: z.expires_at,
                abgelaufen: !!z.expires_at && Date.parse(z.expires_at) < jetzt
            })),
            speech: blase ? { text: blase.text, status: blase.status } : null
        });
    } catch (err) {
        console.error('[shop] me:', err.message);
        res.status(500).json({ error: 'load_failed' });
    }
});

/**
 * POST /api/shop/equip  { skin_key }
 * Nur Figuren-Skins: Sprechblasen nutzen `equipped` bewusst nicht
 * (sonst wuerde der Teil-1-Index nur EINEN aktiven Eintrag erlauben).
 */
router.post('/equip', authenticateViewer, async (req, res) => {
    const key = String(req.body?.skin_key || '');
    if (!key) return res.status(400).json({ error: 'skin_key erforderlich' });

    try {
        const { data: skin } = await supabase
            .from('skins').select('id, kind').eq('key', key).maybeSingle();
        if (!skin) return res.status(404).json({ error: 'unknown_skin' });
        if (skin.kind !== 'skin') return res.status(400).json({ error: 'not_equippable' });

        const { data: besitz } = await supabase
            .from('viewer_skins')
            .select('id, expires_at')
            .eq('viewer_id', req.viewer.id).eq('skin_id', skin.id)
            .is('revoked_at', null)
            .maybeSingle();
        if (!besitz) return res.status(403).json({ error: 'not_owned' });
        if (besitz.expires_at && Date.parse(besitz.expires_at) < Date.now()) {
            return res.status(403).json({ error: 'expired' });
        }

        // Erst alles abwaehlen, dann genau einen setzen — der Teil-1-Index
        // erlaubt nur einen aktiven Eintrag pro Zuschauer.
        await supabase.from('viewer_skins').update({ equipped: false }).eq('viewer_id', req.viewer.id);
        const { error: setzErr } = await supabase.from('viewer_skins')
            .update({ equipped: true }).eq('id', besitz.id);
        if (setzErr) throw setzErr;

        res.json({ ok: true, equipped: key });
    } catch (err) {
        console.error('[shop] equip:', err.message);
        res.status(500).json({ error: 'equip_failed' });
    }
});

/**
 * POST /api/shop/speech  { text }
 * Jeder setzt seinen EIGENEN Text — anders als der Bot-Pfad, bei dem der
 * Streamer im Namen des Zuschauers schreibt.
 */
router.post('/speech', authenticateViewer, async (req, res) => {
    const pruefung = checkSpeechText(req.body?.text);
    if (!pruefung.ok) return res.status(422).json({ error: 'rejected', reason: pruefung.reason });

    try {
        const { data: besitz } = await supabase
            .from('viewer_skins')
            .select('expires_at, skins!inner(kind)')
            .eq('viewer_id', req.viewer.id)
            .eq('skins.kind', 'speech')
            .is('revoked_at', null)
            .maybeSingle();

        if (!besitz) return res.status(403).json({ error: 'not_owned' });
        if (besitz.expires_at && Date.parse(besitz.expires_at) < Date.now()) {
            return res.status(403).json({ error: 'expired' });
        }

        const { error } = await supabase.from('viewer_speech').upsert({
            viewer_id: req.viewer.id,
            text: pruefung.text,
            status: 'approved',
            updated_at: new Date().toISOString()
        }, { onConflict: 'viewer_id' });
        if (error) throw error;

        res.json({ ok: true, text: pruefung.text });
    } catch (err) {
        console.error('[shop] speech:', err.message);
        res.status(500).json({ error: 'save_failed' });
    }
});
```

(function () {

    const BACKEND = 'https://heistbot-production.up.railway.app';
    const SCHLUESSEL_AUSWEIS = 'shop_token';
    const SCHLUESSEL_REF = 'shop_ref';

    function refMerken() {

            ? preis(item.price_cents) + ' · ' + item.duration_days + ' Tage'

    async function katalogLaden() {
        const liste = document.getElementById('katalogListe');
        const status = document.getElementById('katalogStatus');
        if (!liste) return;

        let items = null;
        try {
            const res = await fetch(BACKEND + '/api/shop/catalog');
            if (!res.ok) throw new Error('HTTP ' + res.status);
            items = (await res.json()).items || [];
            localStorage.setItem(SCHLUESSEL_KATALOG, JSON.stringify(items));
        } catch (err) {
            // Kein leerer Bildschirm: letzte bekannte Fassung zeigen.
            const cache = localStorage.getItem(SCHLUESSEL_KATALOG);
            items = cache ? JSON.parse(cache) : [];

                ? 'Zurzeit offline — angezeigt wird der zuletzt bekannte Stand.'
                : 'Der Katalog ist gerade nicht erreichbar. Bitte später noch einmal versuchen.';
        }

        liste.innerHTML = '';
        items.forEach(i => liste.appendChild(karteBauen(i)));
        if (items.length && status) status.hidden = true;
        if (window.lucide) window.lucide.createIcons();
    }

    document.addEventListener('DOMContentLoaded', () => {
        refMerken();
        katalogLaden();
    });
})();

    const RUECKKEHR = location.origin + '/shop/';

    function anmeldeUrl() {
        const p = new URLSearchParams({
            client_id: TWITCH_CLIENT_ID,
            redirect_uri: RUECKKEHR,
            response_type: 'token',
            scope: '',
            force_verify: 'false'

    /** Twitch legt den Token ins URL-Fragment. Einmal einloesen, dann entfernen. */
    async function tokenAusFragmentEinloesen() {
        if (!location.hash.includes('access_token=')) return false;
        const token = new URLSearchParams(location.hash.slice(1)).get('access_token');
        history.replaceState(null, '', location.pathname + location.search);
        if (!token) return false;

        try {
            const res = await fetch(BACKEND + '/api/shop/login', {
                method: 'POST',
                headers: { 'Content-Type': 'application/json' },
                body: JSON.stringify({ twitchAccessToken: token })
            });
            if (!res.ok) throw new Error('HTTP ' + res.status);
            const daten = await res.json();
            localStorage.setItem(SCHLUESSEL_AUSWEIS, daten.token);
            localStorage.setItem('shop_name', daten.viewer.displayName || '');
            return true;
        } catch (err) {
            meldung('Die Anmeldung hat nicht geklappt. Versuch es bitte noch einmal.');
            return false;
        }
    }
